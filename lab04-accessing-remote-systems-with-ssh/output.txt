Task 3 – Secure File Transfer Using SCP
------------------------------------------------------------

toor@ip-172-31-10-91:~$ echo "This is a secure file transfer test." > localfile.txt

------------------------------------------------------------

toor@ip-172-31-10-91:~$ scp localfile.txt student@172.31.10.200:/home/student/

localfile.txt                                  100%   38     8.1KB/s   00:00

------------------------------------------------------------

toor@ip-172-31-10-91:~$ ssh student@172.31.10.200

student@ip-172-31-10-200:~$ ls
localfile.txt

student@ip-172-31-10-200:~$ exit

------------------------------------------------------------

toor@ip-172-31-10-91:~$ ssh student@172.31.10.200
student@ip-172-31-10-200:~$ echo "Remote system file." > remotefile.txt
student@ip-172-31-10-200:~$ exit

------------------------------------------------------------

toor@ip-172-31-10-91:~$ scp student@172.31.10.200:/home/student/remotefile.txt ~/

remotefile.txt                                100%   19     5.2KB/s   00:00

------------------------------------------------------------

toor@ip-172-31-10-91:~$ cat remotefile.txt
Remote system file.
